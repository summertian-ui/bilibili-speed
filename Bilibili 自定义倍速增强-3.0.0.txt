// ==UserScript==
// @name         Bilibili 自定义倍速增强
// @namespace    https://www.bilibili.com/
// @version      4.0.0
// @description  B站增加1.75x 2.5x 3.0x倍速
// @author       老大
// @match        *://*.bilibili.com/*
// @grant        none
// @run-at       document-startå
// ==/UserScript==


(function () {

    'use strict';


    console.log('[B站倍速] 启动');


    const CUSTOM_RATES = [
        3.0,
        2.5,
        1.75
    ];


    const MENU =
        '.bpx-player-ctrl-playbackrate-menu';


    const ITEM =
        '.bpx-player-ctrl-playbackrate-menu-item';



    function getVideo(){

        return document.querySelector(
            '.bpx-player-video-wrap video'
        )
        ||
        document.querySelector('video');

    }



    /**
     * 设置速度
     */
    function setRate(rate){

        let video=getVideo();


        if(!video){

            console.log(
                '[B站倍速] 未找到video'
            );

            return;
        }


        video.playbackRate =
            Number(rate);


        console.log(
            '[B站倍速]',
            rate+'x'
        );


        updateActive(rate);

    }




    /**
     * 高亮
     */
    function updateActive(rate){


        document
        .querySelectorAll(ITEM)
        .forEach(el=>{


            if(
                Number(el.dataset.value)
                ===
                Number(rate)
            ){

                el.classList.add(
                    'bpx-state-active'
                );

            }else{

                el.classList.remove(
                    'bpx-state-active'
                );
            }

        });

    }





    /**
     * 创建菜单
     */
    function createItem(rate){


        let li =
            document.createElement('li');


        li.className =
            'bpx-player-ctrl-playbackrate-menu-item';


        li.dataset.value =
            rate;


        li.dataset.customRate =
            'true';


        li.innerText =
            rate+'x';



        li.onclick=function(e){

            e.stopPropagation();

            setRate(rate);

        };


        return li;

    }





    /**
     * 注入
     */
    function inject(){


        let menu =
            document.querySelector(MENU);


        if(!menu)
            return;



        CUSTOM_RATES.forEach(rate=>{


            // 已存在
            if(
                menu.querySelector(
                    `[data-value="${rate}"]`
                )
            ){
                return;
            }



            let item =
                createItem(rate);



            /*
             * 找插入位置
             *
             * 3.0
             * 2.5
             * 2.0
             * 1.75
             */


            let nodes =
                [...menu.children];


            let insertBefore =
                nodes.find(
                    el =>
                    Number(el.dataset.value)
                    <
                    rate
                );



            if(insertBefore){

                menu.insertBefore(
                    item,
                    insertBefore
                );

            }else{

                menu.appendChild(item);

            }


        });

    }





    /**
     * 监听DOM
     */
    const observer =
        new MutationObserver(()=>{

            inject();

        });



    function start(){


        observer.observe(
            document.documentElement,
            {
                childList:true,
                subtree:true
            }
        );


        inject();


    }




    /**
     * 视频变化监听
     */
    let lastVideo=null;


    setInterval(()=>{


        let video=getVideo();


        if(
            video &&
            video!==lastVideo
        ){

            lastVideo=video;


            console.log(
                '[B站倍速] 新视频'
            );


            setTimeout(
                inject,
                500
            );

        }


    },1000);





    if(
        document.documentElement
    ){

        start();

    }else{


        window.addEventListener(
            'DOMContentLoaded',
            start
        );

    }



})();
